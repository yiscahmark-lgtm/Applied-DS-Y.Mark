import pandas as pd
print("NORMALIZED USDA ALMOND TREE TABLE")

try:
    df_usda = pd.read_csv(usda_path)

    # bearing acreage only
    df_usda_bearing = df_usda[df_usda['Data Item'] == 'ALMONDS - ACRES BEARING'].copy()

    # removing commas and convert to numeric integers
    df_usda_bearing['Value'] = df_usda_bearing['Value'].astype(str).str.replace(',', '')
    df_usda_bearing['Value'] = pd.to_numeric(df_usda_bearing['Value'], errors='coerce')

    # rename tracking columns
    df_usda_clean = df_usda_bearing[['Year', 'Value']].copy()
    df_usda_clean.columns = ['HARVEST_YEAR', 'FRESNO_BEARING_ACRES']
    df_usda_clean = df_usda_clean.sort_values(by='HARVEST_YEAR').reset_index(drop=True)

    display(df_usda_clean.head(10))
except Exception as e:
    print(f"ERROR{e}")

import pandas as pd

print("NORMALIZED NOAA WEATHER DATA")

try:
    # convert strings to date time
    df_temp['DATE'] = pd.to_datetime(df_temp['DATE'])

    # pull temperature column and stripping any text
    if 'HourlyDryBulbTemperature' in df_temp.columns:
        df_temp['HOURLY_TEMP_F'] = pd.to_numeric(df_temp['HourlyDryBulbTemperature'].astype(str).str.extract(r'([0-9.-]+)')[0], errors='coerce')
    else:
        df_temp['HOURLY_TEMP_F'] = pd.to_numeric(df_temp['TMP'].astype(str).str.split(',').str[0], errors='coerce') / 10.0 * 1.8 + 32

    # drop null rows
    df_temp_clean = df_temp[['STATION', 'DATE', 'HOURLY_TEMP_F']].dropna().copy()

    print(f"Cleaned Dataset: {df_temp_clean.shape[0]} hourly records.\n")
    display(df_temp_clean.head(5))
except Exception as e:
    print(f"ERROR{e}")

import pandas as pd
print("NORMALIZED WINTER MONTHS")

# removing the month number from date column to isolate winter month timeframes
df_temp_clean['MONTH'] = df_temp_clean['DATE'].dt.month
df_winter_only = df_temp_clean[df_temp_clean['MONTH'].isin([11, 12, 1, 2])].copy()
df_winter_only = df_winter_only.drop(columns=['MONTH'])

def get_harvest_year(row_date):
    if row_date.month in [11, 12]:
        return row_date.year + 1
    return row_date.year

df_winter_only['HARVEST_YEAR'] = df_winter_only['DATE'].apply(get_harvest_year)
print(f"Winter Hours Remaining: {df_winter_only.shape[0]}")

import pandas as pd

print("\nCLEANED, MERGED DATA")
df_weather_annual = df_winter_only.groupby('HARVEST_YEAR')['HOURLY_TEMP_F'].mean().reset_index()
df_weather_annual.columns = ['HARVEST_YEAR', 'AVERAGE_WINTER_TEMP_F']

df_final_project_data = pd.merge(df_weather_annual, df_usda_clean, on='HARVEST_YEAR', how='inner')

print(f"Rows Matched Between Files: {df_final_project_data.shape[0]}\n")
display(df_final_project_data)
